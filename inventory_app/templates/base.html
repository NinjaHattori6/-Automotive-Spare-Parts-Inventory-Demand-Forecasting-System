from flask import Blueprint, flash, redirect, render_template, request, url_for from sqlalchemy import or_ from inventory_app.extensions import db from inventory_app.models import Product, Supplier Supplier.email.ilike(term), ) ) suppliers = db.session.execute(query).scalars().all() return render_template("suppliers/index.html", suppliers=suppliers, search=search) @bp.route("/new", lead_time_days = int(request.form.get("lead_time_days", 0) or 0) except ValueError: flash("Lead time days must be a valid number.", "danger") return render_template("suppliers/form.html", supplier=None, action="Create") if not supplier_name: flash("Supplier name is required.", "danger") else: supplier = Supplier( supplier_name=supplier_name, contact_person=contact_person, email=email, redirect(url_for("suppliers.index")) return render_template("suppliers/form.html", supplier=None, action="Create") @bp.route("/", methods=["GET"]) @login_required def supplier_detail(supplier_id): supplier = db.session.get(Supplier, supplier_id) if supplier is None: flash("Supplier not found.", "warning") return redirect(url_for("suppliers.index")) products = db.session.execute( request.form.get("address", "").strip() try: supplier.lead_time_days = int(request.form.get("lead_time_days", 0) or 0) except ValueError: flash("Lead time days must be a valid number.", "danger") @bp.route("//delete", methods=["POST"]) @login_required def delete_supplier(supplier_id): supplier = db.session.get(Supplier, supplier_id) if supplier is None: flash("Supplier not found.", "warning")
